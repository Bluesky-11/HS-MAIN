{--- 一 / 二 / 三、周期与参数区 ---}
DAY_PRD    := 5;
IS_DAY_BAR := PERIOD = DAY_PRD;
IS_MIN_BAR := PERIOD >= 0 AND PERIOD <= 4;
IS_HTF     := PERIOD >= 6;

N_TRK      := 10;
ATR_X      := 1.06;
ATR_XS     := 1.48;
V_DYS      := IF(IS_MIN_BAR, 5, 15);
SHR_R      := 0.85;
BB_F       := 2.0;
HOLD_MAX   := 240;
M_PER      := 13;
M_PER2     := 55;
EPS        := 0.0001;
MA13_TOL   := 0.985;
CORE_TOL   := 0.982;
STRONG_GAIN_PCT := 20;
TIGHT_STOP_MULT := 0.68;
MID_STOP_MULT   := 0.88;
MID_GAIN_PCT    := 12;
SZ_SC_MID  := 0.90;
SZ_SC_HI   := 1.30;
SZ_OFF     := 1.9;
SZ_DIV     := 2.7;
SZ_STD_N   := IF(IS_MIN_BAR, 15, 80);
WARM_N     := IF(IS_MIN_BAR, 45, 60);
OPEN_BLOCK := 40;

KC_MULT     := 1.45;
GAP_PCT     := 0.5;

DAMP_MIN    := 60;
K_WICK_W    := 0.1;
K_BODY_W    := 2;

DN_COEF_NRM := 0.85;

{--- V3.3 新增参数(原硬编码) ---}
BODY_SH     := 0.40;     { 实体收缩阈值 }
LOWV_R      := 0.80;     { 极度缩量比(相对V_MA_SAFE) }

FLY_DARK    := -0.02;    { 起飞暗流下限 }
SZ_VST_TH   := 1.15;     { 资金强势阈值 }

EXH_SZ_MIN  := -0.5;     { 衰竭资金底线 }
DIV_N       := 15;       { 背离低点回看周期 }

{==================== 四、公共基础 ====================}
N_BAR   := BARSCOUNT(C);
WARM_OK := N_BAR > WARM_N;

IN_SESS := FROMOPEN > 0 AND FROMOPEN < TRD_MIN;
IS_PRED := ISLASTBAR AND IN_SESS AND IS_DAY_BAR;
PRED_F  := IF(IS_PRED, TRD_MIN / MAX(FROMOPEN, VP_MIN), 1);
V_PRED  := VOL * PRED_F;

PC1     := IF(N_BAR > 1, REF(C, 1), O);
PREV_C  := MAX(PC1, 0.01);
FIRST_C := VALUEWHEN(N_BAR = 1, C);
TR_BASE := MAX(MAX(H - L, ABS(H - PREV_C)), ABS(L - PREV_C));

NET_CHG  := IF(N_BAR > 10, ABS(C - REF(C, 10)), ABS(C - FIRST_C));
PATH_SUM := IF(N_BAR > 10, SUM(ABS(C - PC1), 10), SUM(ABS(C - PC1), 0));
ER_SUB   := IF(PATH_SUM < EPS, 0.5, MIN(NET_CHG / (PATH_SUM + EPS), 1));
W_ER     := EMA(ER_SUB, 5);

TIME_DAMP := IF(IS_PRED AND FROMOPEN < DAMP_MIN, MAX(FROMOPEN, 5) / DAMP_MIN, 1.0);

K_RNG   := MAX(H - L, MAX(C * 0.002, ABS(C - PREV_C)));
IN_DIR  := ((C - L) - (H - C)) / (K_RNG + EPS);
RL_DIR  := IF(ABS(IN_DIR) < 0.01, IF(C > PREV_C, 0.6, IF(C < PREV_C, -0.6, 0)), IN_DIR);

ATR_FAST := MA(TR_BASE, 12);
ATR_MID  := MA(TR_BASE, 24);
ATR_SLOW := MA(TR_BASE, 40);
ATR_VAL  := W_ER * ATR_FAST + (1 - W_ER) * (0.6 * ATR_MID + 0.4 * ATR_SLOW);

MA13     := EMA(C, M_PER);
SYS_MA55 := EMA(C, M_PER2);
MA13_P1  := IF(N_BAR > 1, REF(MA13, 1), MA13);
MA13_P5  := IF(N_BAR > 5, REF(MA13, 5), MA13);
MA55_P1  := IF(N_BAR > 1, REF(SYS_MA55, 1), SYS_MA55);

SLP_13    := (MA13 - MA13_P5) / 5 / (MA13 + EPS);
TREND_ST  := SLP_13 > SLP_TH;
STK_OK    := C > MA13 * MA13_TOL;
SLF_TR_UP := SYS_MA55 >= MA55_P1 OR C > SYS_MA55;

C_BIAS_MA    := (C - MA13) / (MA13 + EPS);
SZ_DOWN_COEF := IF(C_BIAS_MA > BIAS_TH, DN_COEF_UP, DN_COEF_NRM);

TREND_OWN := IF(MA13 > SYS_MA55, 35, 0)
           + IF(MA13 > MA13_P1, 25, IF(MA13 >= MA13_P1 * 0.9999, 10, 0))
           + IF(SYS_MA55 > MA55_P1, 20, 0)
           + IF(C > MA13, 20, 0);
D_C      := C#DAY;
D_M13    := EMA(C, M_PER)#DAY;
D_M55    := EMA(C, M_PER2)#DAY;
D_M13_P  := REF(EMA(C, M_PER), 1)#DAY;
D_M55_P  := REF(EMA(C, M_PER2), 1)#DAY;
TREND_D  := IF(D_M13 > D_M55, 35, 0)
          + IF(D_M13 > D_M13_P, 25, IF(D_M13 >= D_M13_P * 0.9999, 10, 0))
          + IF(D_M55 > D_M55_P, 20, 0)
          + IF(D_C > D_M13, 20, 0);
TREND_DAY     := IF(IS_HTF, TREND_OWN, TREND_D);
MULTI_RESO_OK := TREND_DAY >= 55;

HD := H - IF(N_BAR > 1, REF(H, 1), H);
LD := IF(N_BAR > 1, REF(L, 1), L) - L;
DMP_RAW := IF(HD > 0 AND HD > LD, HD, 0);
DMM_RAW := IF(LD > 0 AND LD > HD, LD, 0);

TR_F  := EMA(TR_BASE, 8);
PDI_F := EMA(DMP_RAW, 8) * 100 / (TR_F + EPS);
MDI_F := EMA(DMM_RAW, 8) * 100 / (TR_F + EPS);
TR_S  := EMA(TR_BASE, 20);
PDI_S := EMA(DMP_RAW, 20) * 100 / (TR_S + EPS);
MDI_S := EMA(DMM_RAW, 20) * 100 / (TR_S + EPS);

ADX_F   := EMA(ABS(MDI_F - PDI_F) / (MDI_F + PDI_F + EPS) * 100, 8);
ADX_S   := EMA(ABS(MDI_S - PDI_S) / (MDI_S + PDI_S + EPS) * 100, 20);
ADX_VAL := W_ER * ADX_F + (1 - W_ER) * ADX_S;
ADX_STR := ADX_VAL > 20;

V_MA      := IF(N_BAR >= V_DYS, MA(V_PRED, V_DYS), SUM(V_PRED, 0) / MAX(N_BAR, 1));
V_MA_SAFE := MAX(V_MA, V_PRED * V_FLOOR_R);
V_SHRK    := V_PRED < V_MA_SAFE * SHR_R;
V_MA5     := IF(N_BAR >= 5, MA(V_PRED, 5), SUM(V_PRED, 0) / MAX(N_BAR, 1));
VOL_RATIO := V_PRED / (V_MA_SAFE + EPS);

{==================== 五、主图资金引擎 ====================}
SZ_DIR     := IF(C >= PREV_C, VOL_RATIO, -VOL_RATIO * SZ_DOWN_COEF);
M_SZ_OSC_F := EMA(SZ_DIR, 5) - EMA(SZ_DIR, 15);
M_SZ_OSC_S := EMA(SZ_DIR, 15) - EMA(SZ_DIR, 45);
M_SZ_OSC   := W_ER * M_SZ_OSC_F + (1 - W_ER) * M_SZ_OSC_S;

SZ_STD_RAW := STD(M_SZ_OSC, SZ_STD_N);
SZ_STD     := IF(SZ_STD_RAW > 0.005, SZ_STD_RAW, 0.02);
SZ         := M_SZ_OSC / SZ_STD;
SZ_P1      := IF(N_BAR > 1, REF(SZ, 1), SZ);

SZ_SC      := (SZ + SZ_OFF) / SZ_DIV;
SZ_ACC_S   := SZ > SZ_P1 AND SZ > SZ_ACC_MIN;
SZ_VST     := SZ_SC > SZ_VST_TH;
SZ_STR     := SZ_SC > SZ_STR_TH;
SCALPEL    := V_SHRK AND SZ_ACC_S;
EX_CAP     := MIN(VOL_RATIO, EX_RATIO_CAP);

X_PDIR     := IF(C > PREV_C, 1, IF(C < PREV_C, -1, 0));
X_FLOW     := IF(X_PDIR * IN_DIR < 0, 0.75, 1.0);
X_RDARK    := RL_DIR * (0.60 + 0.40 * EX_CAP) * X_FLOW;
{ 恒定周期EMA按效率比切换，替代变周期EMA，兼容性更好 }
M_DARK     := IF(W_ER > 0.6, EMA(X_RDARK, 2), EMA(X_RDARK, 5));

M_D1       := (SZ - SZ_P1) * TIME_DAMP;
M_D1_P     := IF(N_BAR > 1, REF(M_D1, 1), M_D1);
M_D2       := M_D1 - M_D1_P;
M_LEAD     := (M_D1 + M_D2 * LEAD_PHI) * LEAD_K;
M_SZ_MA    := EMA(SZ + M_LEAD * LEAD_GAIN, 2);

M_TMC_BASE := W_ER * EMA(SZ, 7) + (1 - W_ER) * EMA(SZ, 25);
M_TMC_P1   := IF(N_BAR > 1, REF(M_TMC_BASE, 1), M_TMC_BASE);
M_TMC_D1   := (M_TMC_BASE - M_TMC_P1) * TIME_DAMP;
M_TMC_D1P  := IF(N_BAR > 1, REF(M_TMC_D1, 1), M_TMC_D1);
M_TMC_D2   := M_TMC_D1 - M_TMC_D1P;
M_TMC_LR   := (M_TMC_D1 + M_TMC_D2 * TMC_PHI) * TMC_K;
M_TMC_LF   := M_TMC_LR * TMC_MIX + EMA(M_TMC_LR, 2) * (1 - TMC_MIX);
M_TMC_OBS  := EMA(M_TMC_BASE + M_TMC_LF, 2);

M_CROSS_DN := CROSS(M_TMC_OBS, M_SZ_MA);
M_ACC_BOOST := MAX(0, M_D2);
EXPL_TH_SUB := 1.85 - W_ER * 0.60 - MIN(0.25, M_ACC_BOOST * 0.5);
IS_EXPL_SUB := V_PRED > V_MA_SAFE * EXPL_TH_SUB;
M_CF_TREND  := M_SZ_MA > M_TMC_OBS;
SZ_STD20    := STD(SZ, 20);
SZ_MA20     := MA(SZ, 20);
M_QUIET     := COUNT((SZ > 0 AND NOT(IS_EXPL_SUB)) OR (SZ_STD20 * 2 / (ABS(SZ_MA20) + 0.01) < QUIET_TH), 3);
M_FLY       := M_CF_TREND AND REF(M_QUIET >= 2, 1) AND SZ > 0 AND IS_EXPL_SUB AND M_DARK > FLY_DARK;

{==================== 六、买入信号与衰竭埋伏判定 (V3.3) ====================}
LLV60 := LLV(L, 60);
HH20  := HHV(H, 20);
HH10C := HHV(C, 10);
HHV3  := HHV(H, 3);
LLV3  := LLV(L, 3);

BODY_SHRINK := ABS(C - O) / (H - L + EPS) < BODY_SH;
LOW_VOL     := V_PRED < V_MA5 * SHR_R OR V_PRED < V_MA_SAFE * LOWV_R;

{【背离重构】当前为DIV_N周期价格低点，且不高于前一个(间隔>=DIV_GAP)低点，
  但资金动能SZ高于前低点时的SZ，同时短线SZ回升 → 真底背离}
PLOW_FLAG   := L = LLV(L, DIV_N);
PLOW_OLD    := VALUEWHEN(REF(PLOW_FLAG, DIV_GAP), REF(L, DIV_GAP));
SZ_OLD      := VALUEWHEN(REF(PLOW_FLAG, DIV_GAP), REF(SZ, DIV_GAP));
SZ_DIVERG   := PLOW_FLAG AND PLOW_OLD > 0 AND L <= PLOW_OLD AND SZ > SZ_OLD AND SZ > REF(SZ, 2);
SZ_REPAIR   := SZ > REF(SZ, 1) AND M_DARK > REF(M_DARK, 1);

{【极致衰竭】形态收敛 + 极度缩量 + 资金基因(剔除死水盘)}
RAW_EXHAUSTION := ((HHV3 - LLV3) / (LLV3 + EPS) * 100 < EXH_RNG OR BODY_SHRINK)
                AND LOW_VOL AND C >= MA13 * CORE_TOL AND C <= MA13 * MA13_UP
                AND (C - LLV60) / (LLV60 + EPS) * 100 < EXH_BASE
                AND (SLP_13 > -0.0002 OR C > MA13)
                AND (SZ_DIVERG OR SZ_REPAIR OR SZ > EXH_SZ_MIN);

EXH_CNT5       := COUNT(RAW_EXHAUSTION, 5);
TRY_EXHAUSTION := RAW_EXHAUSTION AND EXH_CNT5 = 1;

DONCHIAN_HIGH := REF(HHV(H, N_TRK), 1);
BREAK_POWER   := (C - DONCHIAN_HIGH) / (ATR_VAL + EPS) > BRK_PWR;
RAW_BUY       := CROSS(C, DONCHIAN_HIGH) AND BREAK_POWER AND STK_OK;

KC_UP    := MA13 + KC_MULT * ATR_VAL;
BB_UP    := MA13 + BB_F * STD(C, M_PER);
BB_BREAK := CROSS(C, BB_UP) AND V_PRED > REF(V_PRED, 1) * BB_VOL_R;
H_P1     := IF(N_BAR > 1, REF(H, 1), H);
GAP_SZ   := (L - H_P1) / (H_P1 + EPS) * 100;
MAIN_GAP := GAP_SZ > GAP_PCT AND C > O AND V_PRED > V_MA5;

BUY_T    := RAW_BUY OR CROSS(C, KC_UP) OR BB_BREAK OR MAIN_GAP;

OPEN_BLK  := FROMOPEN > 0 AND FROMOPEN <= OPEN_BLOCK;
OPEN_SAFE := NOT(IF(IS_MIN_BAR, (OPEN_HIST = 1 OR ISLASTBAR) AND OPEN_BLK, IS_PRED AND OPEN_BLK));

CORE_COND := BUY_T AND C > MA13 * CORE_TOL AND SLF_TR_UP;
FLT_PASS  := SCALPEL OR MAIN_GAP OR SZ_STR OR (C > HH20 * NEAR_HH AND VOL_RATIO > 1.0) OR (M_DARK > DARK_BUY) OR M_FLY;

FINAL_BUY := CORE_COND AND FLT_PASS AND MULTI_RESO_OK AND OPEN_SAFE AND WARM_OK;
SIG_STR   := FINAL_BUY AND (TREND_ST OR SZ_VST OR BREAK_POWER);
SIG_NOR   := FINAL_BUY AND NOT(SIG_STR);

TRY_BUY_RAW := TRY_EXHAUSTION AND NOT(FINAL_BUY) AND MULTI_RESO_OK AND OPEN_SAFE AND WARM_OK;
BUY_SIG     := SIG_STR OR SIG_NOR;

{==================== 七、棘轮止损与持仓状态 ====================}
HAS_BUY  := SUM(BUY_SIG, 0) > 0;
B_CNT    := IF(HAS_BUY, BARSLAST(BUY_SIG), 9999);
HOLD_LEN := IF(HAS_BUY, MIN(B_CNT, HOLD_MAX) + 1, 1);

BUY_REF_C := REF(C, HOLD_LEN - 1);
HH_C_HOLD := HHV(C, HOLD_LEN);
MAX_GAIN  := IF(HAS_BUY, (HH_C_HOLD - BUY_REF_C) / (BUY_REF_C + EPS) * 100, 0);

DYN_X := IF(MAX_GAIN > STRONG_GAIN_PCT OR SZ_SC > SZ_SC_HI, TIGHT_STOP_MULT,
        IF(MAX_GAIN > MID_GAIN_PCT OR SZ_SC > SZ_SC_MID, MID_STOP_MULT,
        IF(TREND_ST AND ADX_STR, ATR_XS, ATR_X)));
TIGHT_LOCK := DYN_X <= TIGHT_STOP_MULT;

RAW_TRAIL    := HH_C_HOLD - DYN_X * ATR_VAL;
STOP_RATCHET := HHV(RAW_TRAIL, HOLD_LEN);

CRASH_EX   := C < STOP_RATCHET OR CROSS(MA13, C) OR C < MA13 * IF(ADX_STR, CR_ADX, CR_NRM) OR B_CNT >= HOLD_MAX;
LAST_CRASH := IF(SUM(CRASH_EX, 0) > 0, BARSLAST(CRASH_EX), 9999);
HOLD_ST    := HAS_BUY AND LAST_CRASH > B_CNT;

{埋伏仅在空仓时有效，与 IS_WINTER 口径一致}
TRY_BUY    := TRY_BUY_RAW AND NOT(HOLD_ST);

IS_SUMMER := HOLD_ST AND (MAX_GAIN > STRONG_GAIN_PCT OR (C >= HH10C AND SZ_SC < SZ_SC_MID) OR TIGHT_LOCK);
IS_SPRING := HOLD_ST AND NOT(IS_SUMMER);
IS_WINTER := NOT(HOLD_ST) AND WARM_OK AND RAW_EXHAUSTION;
IS_AUTUMN := NOT(HOLD_ST) AND NOT(IS_WINTER);

HOLD_CROSS_DN := HOLD_ST AND M_CROSS_DN;

{==================== 八、进阶：右侧火花确认 ====================}
CONFIRM_BULL := C > O AND C >= PREV_C;
CONFIRM_VOL  := V_PRED > V_MA5 * CF_VOL_R;
CONFIRM_FLG  := REF(IS_WINTER OR TRY_BUY, 1) AND CONFIRM_BULL AND CONFIRM_VOL;

{==================== 九、风险空间与战术节点 ====================}
HH60        := HHV(H, 60);
RWD_SPACE   := MAX(HH60 - C, RR_ATR_MULT * ATR_VAL);
RSK_SPACE   := MAX(IF(HOLD_ST, C - STOP_RATCHET, DYN_X * ATR_VAL), 0.1 * ATR_VAL);
RISK_REWARD := MIN(RWD_SPACE / (RSK_SPACE + EPS), RR_CAP);
D_STOP_VAL  := IF(HOLD_ST, STOP_RATCHET, DRAWNULL);

TN1 := BUY_SIG;
TN2 := NOT(TN1) AND TRY_BUY;
TN3 := NOT(TN1 OR TN2) AND HOLD_CROSS_DN;
TN4 := NOT(TN1 OR TN2 OR TN3) AND HOLD_ST AND TIGHT_LOCK;
TN5 := NOT(TN1 OR TN2 OR TN3 OR TN4) AND HOLD_ST;
TN6 := NOT(TN1 OR TN2 OR TN3 OR TN4 OR TN5);

{==================== 十、视觉渲染与面板 ====================}
P_GAP     := MAX((H - L) * 0.03, C * 0.0005);
PREV_HOLD := IF(N_BAR > 1, REF(HOLD_ST, 1), 0);

SPRING_UP := IS_SPRING AND C >= MA13;
SPRING_DN := IS_SPRING AND C < MA13;
AUTUMN_UP := IS_AUTUMN AND C >= O;
AUTUMN_DN := IS_AUTUMN AND C < O;
