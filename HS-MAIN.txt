{━━━━━━━━━━━━━━━━ 模块 14：衰竭耗尽信号 FRESH_EXH ━━━━━━━━━━━━━━━━}
CONSEC_DRY:=COUNT(VOL_RATIO<0.68,4)>=3;
DF_EXH_OK:=DF_GT_N002 AND (DARK_FLOW>=DF_REF1 OR DARK_FLOW>-0.003);
DF_BOTTOM_DIV:=L<=LLV(L,10) AND DARK_FLOW>REF(LLV(DARK_FLOW,10),1);
RAW_EXH:=WARM_UP AND C>MA21*0.96 AND SZ>-0.9 AND SLP_21>-0.001 AND C<=MA21_UPR_D AND C_DIST_LLV60<45 AND LOW_VOL AND CONSEC_DRY AND (DF_EXH_OK OR DF_BOTTOM_DIV) AND VCP_TIGHT_MA AND (VCP_NORMAL OR VCP_STRONG OR VCP_SUB OR RNG_L3<3.2);
FRESH_EXH:=RAW_EXH AND COUNT(RAW_EXH,8)<=2;

{━━━━━━━━━━━━━━━━ 模块 15：能量综合评分 ENERGY_SCORE ━━━━━━━━━━━━━━━━}
SEC_MOM:=(C/REF(C,10))-(INDEXC/REF(INDEXC,10));
W_S1:=0.18+W_ER*0.22;
W_S3:=0.28-W_ER*0.10;
W_S24:=0.54-W_ER*0.02;
DF_NORM_VAL:=MIN(MAX((DARK_FLOW+0.05)*10,0),1.0);
EXPL_DENOM:=EXPL_TH_DYN-0.8+EPS;

ENG_V1:=MIN(MAX((VOL_RATIO-1.0)/EXPL_DENOM,0),1.0)*W_S1;
ENG_V3:=MIN(VOL_RATIO,3.0)*0.33333*W_S3;
BASE_ENG_SCORE:=(ENG_V1+DF_NORM_VAL*W_S24+ENG_V3)*100;
SEC_MOM_HI:=SEC_MOM>0.01;
SEC_MOM_MID:=SEC_MOM>=-0.01 AND NOT(SEC_MOM_HI);
RS_FACTOR:=1.15*SEC_MOM_HI+1.0*SEC_MOM_MID+0.75*(1-SEC_MOM_HI-SEC_MOM_MID);
ENERGY_SCORE:=MIN(ROUND(BASE_ENG_SCORE*RS_FACTOR),100);
ENG_VALID_OK:=ENERGY_SCORE>=MAIN_ENG_TH;

{━━━━━━━━━━━━━━━━ 模块 15.1：三元共振评分系统 ━━━━━━━━━━━━━━━━}
{ 斜率改为相对值，原写法用绝对价差，阈值对高价股形同虚设 }
MA_SLP_REL:=(MA21-MA21_REF)*I_MA21_E;
MA_SLP_HI:=MA_SLP_REL>MA_SLP_TH;
MA_SLP_MID:=MA_SLP_REL>=0 AND NOT(MA_SLP_HI);
MA_SLP_LO:=MA_SLP_REL>-MA_SLP_TH AND MA_SLP_REL<0;
SCR_MA_DIR:=MA_SLP_HI*30+MA_SLP_MID*20+MA_SLP_LO*10;
SCR_ENERGY:=ENERGY_SCORE*0.35;
SCR_DARK:=DF_NORM_VAL*35;
RESONANCE_SCORE:=ROUND(SCR_MA_DIR+SCR_ENERGY+SCR_DARK);
RESONANCE_OK:=RESONANCE_SCORE>=65;

{━━━━━━━━━━━━━━━━ 模块 16：上方套牢盘真空度 ━━━━━━━━━━━━━━━━}
CHIP_DIST:=(HH60-C)*I_C_E*100;
CHIP_OK:=CHIP_DIST>4.0 OR C>=HH40*0.98;

{━━━━━━━━━━━━━━━━ 模块 17：板块相对强度势能 ━━━━━━━━━━━━━━━━}
INDEX_MA21:=EMA(INDEXC,23);
SECTOR_OK:=SEC_MOM>-0.03 AND (INDEXC>INDEX_MA21 OR INDEXC>=REF(INDEXC,5));

{━━━━━━━━━━━━━━━━ 模块 18：主买入与冬季潜伏信号系统 ━━━━━━━━━━━━━━━━}
DON_HIGH:=REF(HHV(H,N_TRK),1);
BRK_PWR:=(C-DON_HIGH)*I_ATR_E>0.35;
MAIN_GAP:=(L-PREV_H)*I_PH_E*100>0.5 AND C>O AND V_PRED>V_MA5 AND MAVP_OK;
BREAK_DON:=CROSS(C,DON_HIGH) AND BRK_PWR AND STK_OK AND C>O AND VOL_GT095 AND SZ>-0.65 AND DF_GT_N002;
BREAK_KC:=CROSS(C,MA21+KC_MULT*ATR_VAL) AND C>O AND VOL_GT1 AND SZ>-0.55 AND (DARK_FLOW>0 OR SZ_ACC_S);
BREAK_GAP:=MAIN_GAP AND SZ>-0.55 AND DARK_FLOW>-0.03;
BUY_T:=BREAK_DON OR BREAK_KC OR BREAK_GAP;
OPEN_SAFE:=IS_NOT_MIN OR (FROMOPEN>20 OR NOT(ISLASTBAR));
CORE_COND:=BUY_T AND C>MA21_LWR_B AND (MAVP_BULL_ZONE OR MAVP_ADX_STR);

WINTER_RS_OK:=SEC_MOM>=-0.01 OR (INDEXC<=REF(INDEXC,3) AND C>=REF(C,3));

{━━━━━━━━━━━━━━━━ 模块 19：过滤评分系统 FLT_SCORE ━━━━━━━━━━━━━━━━}
FLT_TRIO_RAW:=MIN(CF_FLY_SIG*2+MAVP_ADX_STR*2+STL_TRIG*2,5);
FLT_SZ_PART:=SZ_VST*2+SZ_STR*(1-SZ_VST);
FLT_SCORE:=MIN(SCALPEL+MAIN_GAP*2+FLT_SZ_PART+(C>HH20*0.97 AND VOL_GT1)+(DF_GT008 AND DF_SLOPE_OK)+FLT_TRIO_RAW+SQZ_FIRE_MAIN+IS_EXPL_STRONG+BRK_PWR+SEC_MOM_HI+(WYCKOFF_SPRING*2),FLT_MAX_SCORE);
FLT_PASS:=FLT_SCORE>=FLT_MIN_SCORE;

FINAL_BUY:=WARM_UP AND NOT_LIMIT AND OPEN_SAFE AND MULTI_RESO_OK AND ENG_VALID_OK AND RESONANCE_OK AND CORE_COND AND FLT_PASS AND CHIP_OK AND SECTOR_OK AND (TREND_ST OR SZ_VST OR MAVP_ADX_STR);

TRY_BUY:=WARM_UP AND NOT_LIMIT AND (FRESH_EXH OR STL_TRIG OR WYCKOFF_SPRING) AND NOT(FINAL_BUY) AND MULTI_RESO_OK AND C>=MA21_LWR_C AND (SZ>-0.45 OR DF_GT_N003) AND D1_SM>-0.01 AND DF_SLOPE_OK AND ENERGY_SCORE>=TRY_ENG_TH AND CHIP_OK AND SECTOR_OK AND WINTER_RS_OK;

{━━━━━━━━━━━━━━━━ 模块 20：持仓管理与动态止损系统 ━━━━━━━━━━━━━━━━}
B_CNT_RAW:=BARSLAST(FINAL_BUY);
HAS_TRG:=SUM(FINAL_BUY,0)>0;
B_TRY_RAW:=BARSLAST(TRY_BUY);
HAS_TRY_EV:=SUM(TRY_BUY,0)>0;

B_CNT:=IF(HAS_TRG,B_CNT_RAW,BARS_CNT);
IN_WIN:=HAS_TRG AND B_CNT<=HOLD_MAX;        { 仅在最近一次买点后的持仓窗口内评估出场 }
HOLD_LEN:=IF(HAS_TRG,MIN(MAX(B_CNT+1,HOLD_LEN_MIN),HOLD_MAX),HOLD_LEN_MIN);

BUY_REF_C:=IF(HAS_TRG AND BUY_REF_C_RAW>EPS,BUY_REF_C_RAW,C);
HH_C_HLD:=IF(HAS_TRG,HHV(H,HOLD_LEN),H);    { H 恒不小于 C，无需 MAX(H,C) }
I_BUY_REF_C_E:=1/(BUY_REF_C+EPS);
MAX_GAIN:=IF(HAS_TRG AND BUY_REF_C>EPS,(HH_C_HLD-BUY_REF_C)*I_BUY_REF_C_E*100,0);

ENT_DYN_B:=IF(REF(TREND_ST AND ADX_STR_S,B_CNT),ATR_XS,ATR_X);
GAIN_HI:=MAX_GAIN>STR_GAIN_PCT OR SZ_SC>1.30;
GAIN_MID:=MAX_GAIN>12 OR SZ_SC>SZ_SC_MID;
IS_TREND_EXTEND:=GAIN_HI AND ADX_STR_S AND DI_BULL AND SLP_21>SLP_TH*2;
TIGHT_LCK:=IS_TREND_EXTEND;

{ GAIN_HI 蕴含 GAIN_MID，IS_TREND_EXTEND 蕴含 GAIN_HI，四项互斥，原式冗余因子已合并 }
DYN_X:=TIGT_STP_MULT*IS_TREND_EXTEND+0.80*GAIN_HI*(1-IS_TREND_EXTEND)+0.88*GAIN_MID*(1-GAIN_HI)+ENT_DYN_B*(1-GAIN_MID);

ATR_STP_DIST:=MAX(DYN_X*ATR_VAL,C*STP_MIN_PCT);
HH_C_MINUS_STP:=HH_C_HLD-ATR_STP_DIST;
STOP_RAT_ATR:=IF(HAS_TRG,HHV(HH_C_MINUS_STP,HOLD_LEN),0);
STOP_CHANDELIER:=IF(HAS_TRG,HH_C_HLD-IF(TIGHT_LCK,ATR_VAL_TIGHT,ATR_VAL3),0);
STOP_RAT_CALC:=IF(TIGHT_LCK,MIN(STOP_RAT_ATR,STOP_CHANDELIER),MAX(STOP_RAT_ATR,STOP_CHANDELIER));
STOP_FLOOR:=MAX(BUY_REF_C*STP_FLOOR_PCT,C*STP_FLOOR_C);
STOP_RAT_RAW:=IF(HAS_TRG,MAX(STOP_RAT_CALC,STOP_FLOOR),0);   { 内部用 0 代替 DRAWNULL }
STOP_RAT_PREV:=REF(STOP_RAT_RAW,1);
PREV_IS_VALID:=STOP_RAT_PREV>EPS AND STOP_RAT_PREV<C*(1+ATR_PCT*8) AND NOT(FINAL_BUY);
STOP_RAT:=IF(HAS_TRG,IF(PREV_IS_VALID,MAX(STOP_RAT_RAW,STOP_RAT_PREV),STOP_RAT_RAW),0);

B_TRY_CNT:=IF(HAS_TRY_EV,B_TRY_RAW,BARS_CNT);
TRY_ENTRY_C:=IF(HAS_TRY_EV,REF(C,B_TRY_CNT),C);
STOP_TRY_RAW:=IF(HAS_TRY_EV AND TRY_ENTRY_C>EPS,TRY_ENTRY_C-TRY_STP_MULT*REF(ATR_VAL,B_TRY_CNT),0);
HAS_TRY_ACTIVE:=HAS_TRY_EV AND B_TRY_CNT<B_CNT AND B_TRY_CNT<HOLD_MAX;

ACCEL_DF_WEAK:=DARK_FLOW<REF(DARK_FLOW,ACCEL_COOL_N) OR DARK_FLOW<-0.05;
ACCEL_WARNING:=(ACCEL_CONSEC AND ACCEL_DF_WEAK) OR MORN_TRP;

{ 出场只在持仓窗口内评估；时间止损仅在第 HOLD_MAX 根触发一次 }
EXIT_TRG:=C<STOP_RAT OR (IS_NOT_MIN AND MA_DEATH_CROSS) OR C<MA21_STOP_TH OR ACCEL_WARNING;

{ 冷却期从"首次跌破"起算，避免陈旧止损线持续为真而长期屏蔽新买点 }
CRASH_EDGE:=CRASH_EX AND NOT(REF(CRASH_EX,1));

D_STOP_VAL:=IF(HOLD_ST,STOP_RAT,DRAWNULL);
FINAL_BUY_G:=FINAL_BUY AND EXIT_COOL;
TRY_BUY_G:=TRY_BUY AND EXIT_COOL;
