VCP_TRAP_HL:=MAX(4.5,ATR_PCT*180);
VCP_NO_TRAP:=(H-MAX(O,C))/C_E*100<VCP_TRAP_UPW AND HL_DIFF/PC_E*100<VCP_TRAP_HL;

ADAPTIVE_SHR:=MIN(MIN(MAX(SHR_R*(1+(ATR_PCT-0.03)*2),0.70),0.95)
              *(1+ATR_HI*0.15-NOT(ATR_HI AND ATR_MID)*0.10),0.98);
V_DRY_EXTREME:=V_PRED<V_MA5*0.75 AND VOL_RATIO<ADAPTIVE_SHR;
LOW_VOL:=V_DRY_EXTREME OR (V_SHRK AND V_DRY_UP);

VCP_TIGHT_MA:=ABS(C-MA21)/MA21_E*100<1.2 AND ATR_PCT<0.04;
VCP_TREND_OK:=MA21_RISING AND MA21_UP5 AND C>MA21 AND TREND_DAY>=40;
BOTTOM_RISING:=LL5>LL10*0.995 AND LL10>LL20*0.99;
BOTTOM_SLOPE:=(LL5-LL20)/LL20_E*100>-0.5 AND LL5>LL20*0.98;

VCP_STAGED:=(RNG_L1*VCP_STAGE_TOL>RNG_L2) AND (RNG_L2*VCP_STAGE_TOL>RNG_L3)
            AND RNG_L3<3.8 AND VCP_TREND_OK AND BOTTOM_RISING
            AND HS_DECREASE_OK AND BOTTOM_SLOPE;
VCP_BRK_VOL_OK:=VOL_RATIO>VCP_BRK_VOL_TH;

VCP_STRONG:=VCP_STAGED AND V_DRY_EXTREME AND C>MA21
            AND VCP_TIGHT_MA AND VCP_ACTIVE_VOL AND VCP_NO_TRAP
            AND CONSEC_SHRK AND VCP_BRK_VOL_OK;
VCP_NORMAL:=RNG_L3<4.2 AND LOW_VOL AND NOT(VCP_STRONG)
            AND C>=MA21_LWR_A AND VCP_TIGHT_MA AND VCP_ACTIVE_VOL
            AND VCP_NO_TRAP AND VCP_TREND_OK;
VCP_SUB:=RNG_L3<4.8 AND LOW_VOL AND C>=MA21_LWR_A AND VCP_TREND_OK AND VCP_NO_TRAP;

{━━━━━━━━━━━━━━━━ 模块 14：衰竭耗尽信号 FRESH_EXH ━━━━━━━━━━━━━━━━}
{优化：CONSEC_DRY 复用 VOL_RATIO，DF_EXH_OK 布尔合并}

{━━━━━━━━━━━━━━━━ 模块 15：能量综合评分 ENERGY_SCORE ━━━━━━━━━━━━━━━━}
{优化：W_S 权重系数在同一行展开；DF_NORM 和 EXPL_DENOM 预算后复用}
SEC_MOM:=(C/REF(C,10))-(INDEXC/REF(INDEXC,10));
W_S1:=0.18+W_ER*0.22;
W_S2:=0.38-W_ER*0.16;
W_S3:=0.28-W_ER*0.10;
W_S4:=0.16+W_ER*0.14;
DF_NORM_VAL:=MIN(MAX((DARK_FLOW+0.05)/0.10,0),1.0);
EXPL_DENOM:=EXPL_TH_DYN-0.8+EPS;

ENG_V1:=MIN(MAX((VOL_RATIO-1.0)/EXPL_DENOM,0),1.0)*W_S1;
ENG_V3:=MIN(MAX(MIN(VOL_RATIO,3.0)/3.0,0),1.0)*W_S3;
BASE_ENG_SCORE:=(ENG_V1+DF_NORM_VAL*(W_S2+W_S4)+ENG_V3)*100;
RS_FACTOR:=IF(SEC_MOM>0.01,1.15,IF(SEC_MOM>-0.01,1.0,0.75));
ENERGY_SCORE:=MIN(ROUND(BASE_ENG_SCORE*RS_FACTOR),100);
ENG_VALID_OK:=ENERGY_SCORE>=MAIN_ENG_TH;

{━━━━━━━━━━━━━━━━ 模块 15.1：三元共振评分系统 ━━━━━━━━━━━━━━━━}
{优化：SCR_MA_DIR 三档布尔乘法代替 IF 嵌套}

           +(MA_SLP_VAL>=0 AND MA_SLP_VAL<=0.0005)*20
           +(MA_SLP_VAL>-0.0005 AND MA_SLP_VAL<0)*10;

SCR_DARK:=DF_NORM_VAL*35;

{━━━━━━━━━━━━━━━━ 模块 18：主买入信号系统 ━━━━━━━━━━━━━━━━}

           AND VOL_RATIO>0.95 AND SZ>-0.65 AND DF_GT_N002;

          AND VOL_RATIO>1.00 AND SZ>-0.55 AND (DARK_FLOW>0 OR SZ_ACC_S);

{━━━━━━━━━━━━━━━━ 模块 19：过滤评分系统 FLT_SCORE ━━━━━━━━━━━━━━━━}

         AND (FRESH_EXH OR STL_TRIG) AND NOT(FINAL_BUY)

         AND ENERGY_SCORE>=TRY_ENG_TH AND CHIP_OK AND SECTOR_OK;

                    IF(TIGHT_LCK,HH_C_HLD-ATR_VAL_TIGHT,HH_C_HLD-ATR_VAL3),0);

             IF(NOT(PREV_IS_VALID),STOP_RAT_RAW,MAX(STOP_RAT_RAW,STOP_RAT_PREV)),
